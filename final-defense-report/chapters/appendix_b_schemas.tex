\section{PostgreSQL Schema}

\begin{lstlisting}[language=SQL, caption={Complete database schema}]
-- Users table (authentication)
CREATE TABLE IF NOT EXISTS users (
    id SERIAL PRIMARY KEY,
    email VARCHAR(255) UNIQUE NOT NULL,
    hashed_password VARCHAR(255),
    full_name VARCHAR(255),
    role VARCHAR(50) DEFAULT 'viewer',
    provider VARCHAR(50) DEFAULT 'local',
    provider_id VARCHAR(255),
    created_at TIMESTAMP DEFAULT NOW()
);

-- Pipeline execution history
CREATE TABLE IF NOT EXISTS pipeline_runs (
    id SERIAL PRIMARY KEY,
    run_id VARCHAR(255) UNIQUE NOT NULL,
    status VARCHAR(50) DEFAULT 'running',
    started_at TIMESTAMP DEFAULT NOW(),
    completed_at TIMESTAMP,
    records_processed INTEGER DEFAULT 0,
    error_message TEXT
);

-- Dataset registry (data lake metadata)
CREATE TABLE IF NOT EXISTS dataset_registry (
    id SERIAL PRIMARY KEY,
    run_id VARCHAR(255) REFERENCES pipeline_runs(run_id),
    dataset_name VARCHAR(255) NOT NULL,
    layer VARCHAR(50) NOT NULL,
    s3_path VARCHAR(500),
    record_count INTEGER,
    checksum VARCHAR(64),
    created_at TIMESTAMP DEFAULT NOW()
);

-- Model registry
CREATE TABLE IF NOT EXISTS model_registry (
    id SERIAL PRIMARY KEY,
    run_id VARCHAR(255) REFERENCES pipeline_runs(run_id),
    model_name VARCHAR(255) NOT NULL,
    model_version VARCHAR(50),
    algorithm VARCHAR(100),
    features_used JSONB,
    performance_metrics JSONB,
    created_at TIMESTAMP DEFAULT NOW()
);

-- SLA risk predictions
CREATE TABLE IF NOT EXISTS sla_risk_scores (
    id SERIAL PRIMARY KEY,
    run_id VARCHAR(255) REFERENCES pipeline_runs(run_id),
    region VARCHAR(100),
    score FLOAT NOT NULL,
    explanation JSONB,
    model_version VARCHAR(50),
    created_at TIMESTAMP DEFAULT NOW()
);

-- OSS network anomalies
CREATE TABLE IF NOT EXISTS anomalies (
    id SERIAL PRIMARY KEY,
    run_id VARCHAR(255) REFERENCES pipeline_runs(run_id),
    cell_id VARCHAR(100),
    severity FLOAT,
    anomaly_type VARCHAR(100),
    kpi_values JSONB,
    model_version VARCHAR(50),
    created_at TIMESTAMP DEFAULT NOW()
);

-- BSS revenue anomalies
CREATE TABLE IF NOT EXISTS revenue_anomalies (
    id SERIAL PRIMARY KEY,
    run_id VARCHAR(255) REFERENCES pipeline_runs(run_id),
    subscriber_id VARCHAR(100),
    severity FLOAT,
    anomaly_type VARCHAR(100),
    metrics JSONB,
    model_version VARCHAR(50),
    created_at TIMESTAMP DEFAULT NOW()
);

-- OSS-BSS correlation insights
CREATE TABLE IF NOT EXISTS correlation_insights (
    id SERIAL PRIMARY KEY,
    run_id VARCHAR(255) REFERENCES pipeline_runs(run_id),
    oss_metric VARCHAR(100),
    bss_metric VARCHAR(100),
    corr_value FLOAT,
    corr_type VARCHAR(50),
    p_value FLOAT,
    created_at TIMESTAMP DEFAULT NOW()
);
\end{lstlisting}
