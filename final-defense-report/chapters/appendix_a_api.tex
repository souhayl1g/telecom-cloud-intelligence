\section{Example API Usage}

\begin{lstlisting}[language=bash, caption={Authentication and data retrieval}]
# 1. Get JWT token
TOKEN=$(curl -s -X POST http://localhost:8002/auth/login \
  -H "Content-Type: application/json" \
  -d '{"email":"user@example.com","password":"pass"}' \
  | jq -r '.access_token')

# 2. Query SLA risk
curl -H "Authorization: Bearer $TOKEN" \
  http://localhost:8000/sla-risk

# 3. Query anomalies
curl -H "Authorization: Bearer $TOKEN" \
  http://localhost:8000/anomalies?limit=50

# 4. Query correlations
curl -H "Authorization: Bearer $TOKEN" \
  http://localhost:8000/correlation?limit=20
\end{lstlisting}
